Função anônima tradicional x arrow function

O mesmo evento de clique escrito das duas formas (equivalentes neste caso):

// Função anônima tradicional
botao.addEventListener('click', function () {
  excluirPorId(id);
});

// Arrow function (lambda)
botao.addEventListener('click', () => excluirPorId(id));

Com um parâmetro e retorno implícito:

// Tradicional
const buscarPorId = function (id) {
  return funcionarios.find(function (f) { return f.id === id; });
};

// Arrow function
const buscarPorId = (id) => funcionarios.find((f) => f.id === id);
